// Copy into internal/service only in an isolated checkout of the deployed revision.
package service

import (
	"context"
	"crypto/sha256"
	"encoding/hex"
	"encoding/json"
	"fmt"
	"io"
	"os"
	"strings"
	"time"
)

type ArchiveRepairLine struct {
	ArchiveID string `json:"archive_id"`
	Event     string `json:"event"`
	Start     int64  `json:"start"`
	Length    int64  `json:"length"`
	End       int64  `json:"end"`
	SHA256    string `json:"sha256"`
}

// ReadRepairLine refuses changed bytes, wrong identity, or shifted source offsets.
func ReadRepairLine(file string, ref ArchiveRepairLine) ([]byte, error) {
	if ref.Start < 0 || ref.Length <= 0 || ref.Length > 6*512*1024*1024+8*1024*1024 || ref.End != ref.Start+ref.Length {
		return nil, fmt.Errorf("invalid repair bounds")
	}
	f, err := os.Open(file)
	if err != nil {
		return nil, err
	}
	defer f.Close()
	b := make([]byte, ref.Length)
	if _, err = f.ReadAt(b, ref.Start); err != nil && err != io.EOF {
		return nil, err
	}
	sum := sha256.Sum256(b)
	if hex.EncodeToString(sum[:]) != ref.SHA256 {
		return nil, fmt.Errorf("repair hash mismatch")
	}
	var e tokenAnalysisArchiveEvent
	if err = json.Unmarshal(b, &e); err != nil {
		return nil, err
	}
	if e.ArchiveID != ref.ArchiveID || e.Event != ref.Event {
		return nil, fmt.Errorf("repair identity mismatch")
	}
	return b, nil
}

// InspectRepairLine uses the deployed parser and input rules without writing.
func (s *TokenAnalysisService) InspectRepairLine(b []byte) (*TokenAnalysisResponseFacts, bool, error) {
	var e tokenAnalysisArchiveEvent
	if err := json.Unmarshal(b, &e); err != nil {
		return nil, false, err
	}
	tm, err := time.Parse(time.RFC3339Nano, e.Timestamp)
	if err != nil {
		return nil, false, err
	}
	if e.Event == "response" {
		return &TokenAnalysisResponseFacts{ArchiveID: sanitizeTokenAnalysisText(e.ArchiveID), CompletedAt: tm, Status: e.Status, OutputTokens: tokenAnalysisArchiveOutputTokens(e.Usage), ResponseID: sanitizeTokenAnalysisText(strings.TrimSpace(e.ResponseID)), Forward: validTokenAnalysisForwardFacts(e.TokenAnalysis)}, false, nil
	}
	if e.Event != "request" {
		return nil, false, fmt.Errorf("unsupported repair event")
	}
	body, err := s.repairBody(e)
	if err != nil {
		return nil, false, err
	}
	return nil, tokenAnalysisInputStoreMaxChars(s.cfg) > 0 && strings.TrimSpace(body.LastUserText) != "", nil
}

func (s *TokenAnalysisService) repairBody(e tokenAnalysisArchiveEvent) (TokenAnalysisBodySummary, error) {
	root, err := parseTokenAnalysisRequestBody([]byte(e.Body))
	if err == nil {
		return summarizeTokenAnalysisRoot(e.Endpoint, root, tokenAnalysisMaxPreviewChars(s.cfg)), nil
	}
	if body, ok := SummarizeTokenAnalysisMultipartRequest(e.Endpoint, e.Headers.ContentType, []byte(e.Body), tokenAnalysisMaxPreviewChars(s.cfg)); ok {
		return body, nil
	}
	if e.BodyTruncated {
		return TokenAnalysisBodySummary{SummaryJSON: map[string]any{"degraded": "body_truncated"}}, nil
	}
	return TokenAnalysisBodySummary{}, fmt.Errorf("parse request body: %w", err)
}

// ApplyRepairLine never updates ordinary index watermarks or historical counters.
// inputOnly repairs a partial request write without changing its existing summary.
func (s *TokenAnalysisService) ApplyRepairLine(ctx context.Context, file string, ref ArchiveRepairLine, b []byte, inputOnly bool) error {
	if inputOnly {
		var e tokenAnalysisArchiveEvent
		if err := json.Unmarshal(b, &e); err != nil {
			return err
		}
		tm, err := time.Parse(time.RFC3339Nano, e.Timestamp)
		if err != nil {
			return err
		}
		body, err := s.repairBody(e)
		if err != nil {
			return err
		}
		// A no-input row is deliberately a no-op, never a deletion during recovery.
		if strings.TrimSpace(body.LastUserText) == "" {
			return nil
		}
		return s.storeTokenAnalysisUserInput(ctx, e, tm, body, tokenAnalysisInputStoreMaxChars(s.cfg))
	}
	roots, err := newTokenAnalysisProjectRoots(ctx, s.repo)
	if err != nil {
		return err
	}
	_, _, failed, id, err := s.indexArchiveLine(ctx, file, ref.End, b, false, roots)
	if err != nil {
		return err
	}
	if failed != 0 || id != ref.ArchiveID {
		return fmt.Errorf("repair line was not accepted")
	}
	return roots.flush(ctx, s.repo)
}

// LinkRepairRows reuses normal billing association, but only for repaired IDs.
// It never creates usage_logs, sends gateway requests, or charges an account.
func (s *TokenAnalysisService) LinkRepairRows(ctx context.Context, ids map[string]bool, start, end time.Time) error {
	if err := s.repo.MergePendingResponses(ctx); err != nil {
		return err
	}
	for round := 0; round < tokenAnalysisUsageLinkMaxRounds; round++ {
		var after int64
		released := false
		for {
			batch, err := s.repo.ListUsageLinkCandidates(ctx, start, end, after, tokenAnalysisUsageLinkBatch)
			if err != nil {
				return err
			}
			for _, c := range batch {
				after = c.ID
				if !ids[c.ArchiveID] {
					continue
				}
				freed, err := s.linkUsageCandidate(ctx, c, tokenAnalysisUsageMatchWindow(s.cfg))
				if err != nil {
					return err
				}
				released = released || freed
			}
			if len(batch) < tokenAnalysisUsageLinkBatch {
				break
			}
		}
		if !released {
			return nil
		}
	}
	return nil
}
