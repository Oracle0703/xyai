// Copy into cmd/token-repair/main.go in the isolated deployed checkout.
package main

import (
	"context"
	"database/sql"
	"encoding/json"
	"flag"
	"fmt"
	"os"
	"path/filepath"
	"reflect"
	"time"

	"github.com/Wei-Shaw/sub2api/internal/config"
	"github.com/Wei-Shaw/sub2api/internal/repository"
	"github.com/Wei-Shaw/sub2api/internal/service"
	_ "github.com/lib/pq"
)

func main() {
	if err := run(); err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}
func run() error {
	apply := flag.Bool("apply", false, "apply reviewed missing rows; default is read-only audit")
	root := flag.String("root", "/var/tmp/token-analysis-repair-20261102", "private evidence directory")
	flag.Parse()
	cfg, err := config.LoadForBootstrap()
	if err != nil {
		return err
	}
	dsn := cfg.Database.DSN() + " application_name=token_analysis_repair statement_timeout=30000 lock_timeout=3000"
	if !*apply {
		dsn += " default_transaction_read_only=on"
	}
	db, err := sql.Open("postgres", dsn)
	if err != nil {
		return err
	}
	defer db.Close()
	db.SetMaxOpenConns(2)
	ctx, cancel := context.WithTimeout(context.Background(), 20*time.Minute)
	defer cancel()
	if err = db.QueryRowContext(ctx, "SELECT value FROM security_secrets WHERE key='token_analysis_fingerprint_secret'").Scan(&cfg.TokenAnalysis.FingerprintSecret); err != nil {
		return err
	}
	if len(cfg.TokenAnalysis.FingerprintSecret) < 32 {
		return fmt.Errorf("missing fingerprint secret")
	}
	svc := service.NewTokenAnalysisService(repository.NewTokenAnalysisRepository(db), cfg, nil)
	var plan struct {
		Path    string                      `json:"path"`
		Size    int64                       `json:"size"`
		MtimeNS int64                       `json:"mtime_ns"`
		Rows    []service.ArchiveRepairLine `json:"rows"`
	}
	b, err := os.ReadFile(filepath.Join(*root, "plan.json"))
	if err != nil {
		return err
	}
	if err = json.Unmarshal(b, &plan); err != nil {
		return err
	}
	st, err := os.Stat(plan.Path)
	if err != nil {
		return err
	}
	if st.Size() != plan.Size || st.ModTime().UnixNano() != plan.MtimeNS {
		return fmt.Errorf("source changed since scan")
	}
	var lock *sql.Conn
	if *apply {
		lock, err = db.Conn(ctx)
		if err != nil {
			return err
		}
		defer lock.Close()
		var acquired bool
		if err = lock.QueryRowContext(ctx, "SELECT pg_try_advisory_lock(hashtext('token-analysis-repair:2026-10-31'))").Scan(&acquired); err != nil {
			return err
		}
		if !acquired {
			return fmt.Errorf("repair already running")
		}
		defer lock.ExecContext(context.Background(), "SELECT pg_advisory_unlock(hashtext('token-analysis-repair:2026-10-31'))")
	}
	mode := "audit"
	if *apply {
		mode = "apply"
	}
	journal, err := os.OpenFile(filepath.Join(*root, mode+"-"+time.Now().Format("20060102-150405")+".jsonl"), os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0600)
	if err != nil {
		return err
	}
	defer journal.Close()
	enc := json.NewEncoder(journal)
	counts := map[string]int{}
	fmt.Printf("mode=%s plan_rows=%d input_max_chars=%d preview_chars=%d\n", mode, len(plan.Rows), cfg.TokenAnalysis.InputStoreMaxChars, cfg.TokenAnalysis.MaxPreviewChars)
	for _, ref := range plan.Rows {
		raw, err := service.ReadRepairLine(plan.Path, ref)
		if err != nil {
			return fmt.Errorf("%s: %w", ref.ArchiveID, err)
		}
		facts, inputExpected, err := svc.InspectRepairLine(raw)
		if err != nil {
			return fmt.Errorf("%s: %w", ref.ArchiveID, err)
		}
		var exists, hasInput bool
		if err = db.QueryRowContext(ctx, "SELECT EXISTS(SELECT 1 FROM token_analysis_request_summaries WHERE archive_id=$1), EXISTS(SELECT 1 FROM token_analysis_user_inputs WHERE archive_id=$1)", ref.ArchiveID).Scan(&exists, &hasInput); err != nil {
			return err
		}
		action := "skip"
		inputOnly := false
		if ref.Event == "request" {
			if !exists {
				action = "request"
			} else if !hasInput && inputExpected {
				action = "input"
				inputOnly = true
			}
		} else {
			same, present, err := responsePresent(ctx, db, *facts)
			if err != nil {
				return err
			}
			if present && !same {
				return fmt.Errorf("conflicting completed response: %s", ref.ArchiveID)
			}
			if !same {
				action = "response"
			}
		}
		counts[action]++
		record := map[string]any{"archive_id": ref.ArchiveID, "event": ref.Event, "start": ref.Start, "action": action}
		if *apply && action != "skip" {
			// Save existing derived records before the first mutation; never print content.
			var backup string
			if err = db.QueryRowContext(ctx, `SELECT json_build_object('summary',(SELECT row_to_json(s) FROM token_analysis_request_summaries s WHERE archive_id=$1),'input',(SELECT row_to_json(i) FROM token_analysis_user_inputs i WHERE archive_id=$1),'pending',(SELECT row_to_json(p) FROM token_analysis_pending_responses p WHERE archive_id=$1))::text`, ref.ArchiveID).Scan(&backup); err != nil {
				return err
			}
			record["before"] = json.RawMessage(backup)
			if err = enc.Encode(record); err != nil {
				return err
			}
			if err = journal.Sync(); err != nil {
				return err
			}
			if err = svc.ApplyRepairLine(ctx, plan.Path, ref, raw, inputOnly); err != nil {
				return fmt.Errorf("apply %s %s: %w", action, ref.ArchiveID, err)
			}
			if err = enc.Encode(map[string]any{"archive_id": ref.ArchiveID, "event": ref.Event, "applied": action}); err != nil {
				return err
			}
		} else {
			if err = enc.Encode(record); err != nil {
				return err
			}
		}
	}
	if err = journal.Sync(); err != nil {
		return err
	}
	if *apply {
		ids := map[string]bool{}
		for _, ref := range plan.Rows {
			ids[ref.ArchiveID] = true
		}
		loc, err := time.LoadLocation("Asia/Shanghai")
		if err != nil {
			return err
		}
		start := time.Date(2026, 10, 31, 0, 0, 0, 0, loc)
		if err = svc.LinkRepairRows(ctx, ids, start, start.AddDate(0, 0, 1)); err != nil {
			return fmt.Errorf("repair usage links: %w", err)
		}
	}
	result, _ := json.Marshal(counts)
	fmt.Println(string(result))
	return nil
}

func responsePresent(ctx context.Context, db *sql.DB, want service.TokenAnalysisResponseFacts) (bool, bool, error) {
	got := service.TokenAnalysisResponseFacts{ArchiveID: want.ArchiveID}
	var completed sql.NullTime
	var status, output sql.NullInt64
	err := db.QueryRowContext(ctx, `SELECT completed_at,response_status,response_output_tokens,response_id,forward_path,cache_key_handling,upstream_cache_key_hash,usage_cache_source,upstream_items_hash,upstream_root_hash FROM token_analysis_request_summaries WHERE archive_id=$1
 UNION ALL SELECT completed_at,response_status,response_output_tokens,response_id,forward_path,cache_key_handling,upstream_cache_key_hash,usage_cache_source,upstream_items_hash,upstream_root_hash FROM token_analysis_pending_responses WHERE archive_id=$1 AND NOT EXISTS(SELECT 1 FROM token_analysis_request_summaries WHERE archive_id=$1 AND completed_at IS NOT NULL) ORDER BY completed_at DESC NULLS LAST LIMIT 1`, want.ArchiveID).Scan(&completed, &status, &output, &got.ResponseID, &got.Forward.ForwardPath, &got.Forward.CacheKeyHandling, &got.Forward.UpstreamCacheKeyHash, &got.Forward.UsageCacheSource, &got.Forward.UpstreamItemsHash, &got.Forward.UpstreamRootHash)
	if err == sql.ErrNoRows {
		return false, false, nil
	}
	if err != nil {
		return false, false, err
	}
	if !completed.Valid {
		return false, false, nil
	}
	got.CompletedAt = completed.Time
	if status.Valid {
		got.Status = int(status.Int64)
	}
	if output.Valid {
		v := output.Int64
		got.OutputTokens = &v
	}
	// PostgreSQL timestamptz retains microseconds; lib/pq/server may round submicroseconds.
	delta := got.CompletedAt.Sub(want.CompletedAt)
	if delta < 0 {
		delta = -delta
	}
	got.CompletedAt = want.CompletedAt
	return delta <= time.Microsecond && reflect.DeepEqual(got, want), true, nil
}
