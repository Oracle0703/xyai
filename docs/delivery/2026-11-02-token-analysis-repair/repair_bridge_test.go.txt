func repairFixture(t *testing.T) (string, ArchiveRepairLine, []byte) {
	t.Helper()
	b := []byte(`{"archive_id":"repair-1","event":"request","timestamp":"2026-10-31T01:00:00Z","endpoint":"/v1/responses","body":"{\"model\":\"gpt-test\",\"input\":\"hello\"}"}` + "\n")
	file := filepath.Join(t.TempDir(), "2026-10-31.jsonl")
	prefix := []byte("earlier line\n")
	require.NoError(t, os.WriteFile(file, append(prefix, b...), 0600))
	sum := sha256.Sum256(b)
	return file, ArchiveRepairLine{ArchiveID: "repair-1", Event: "request", Start: int64(len(prefix)), Length: int64(len(b)), End: int64(len(prefix) + len(b)), SHA256: hex.EncodeToString(sum[:])}, b
}

func TestRepairReaderRejectsDriftAndWrongIdentity(t *testing.T) {
	file, ref, want := repairFixture(t)
	got, err := ReadRepairLine(file, ref)
	require.NoError(t, err)
	require.Equal(t, want, got)
	wrong := ref
	wrong.ArchiveID = "other"
	_, err = ReadRepairLine(file, wrong)
	require.ErrorContains(t, err, "identity")
	wrong = ref
	wrong.End++
	_, err = ReadRepairLine(file, wrong)
	require.ErrorContains(t, err, "bounds")
	f, err := os.OpenFile(file, os.O_WRONLY, 0600)
	require.NoError(t, err)
	_, err = f.WriteAt([]byte("X"), ref.Start+10)
	require.NoError(t, err)
	require.NoError(t, f.Close())
	_, err = ReadRepairLine(file, ref)
	require.ErrorContains(t, err, "hash")
}

type repairInputFailureRepo struct {
	tokenAnalysisRepoStub
	fail bool
}

func (r *repairInputFailureRepo) UpsertUserInput(ctx context.Context, in *TokenAnalysisUserInput) error {
	if r.fail {
		return errors.New("injected disk full")
	}
	return r.tokenAnalysisRepoStub.UpsertUserInput(ctx, in)
}

func TestRepairPartialWriteResumesWithoutRewritingSummaryOrWatermark(t *testing.T) {
	file, ref, b := repairFixture(t)
	r := &repairInputFailureRepo{fail: true}
	s := NewTokenAnalysisService(r, &config.Config{TokenAnalysis: config.TokenAnalysisConfig{InputStoreMaxChars: 8000, MaxPreviewChars: 300}}, nil)
	_, expected, err := s.InspectRepairLine(b)
	require.NoError(t, err)
	require.True(t, expected)
	require.Empty(t, r.upserts)
	err = s.ApplyRepairLine(context.Background(), file, ref, b, false)
	require.ErrorContains(t, err, "disk full")
	require.Len(t, r.upserts, 1)
	require.Empty(t, r.userInputs)
	require.Empty(t, r.states)
	require.Equal(t, ref.Start, *r.upserts[0].SourceLineStart)
	require.Equal(t, ref.End, *r.upserts[0].SourceOffset)
	r.fail = false
	require.NoError(t, s.ApplyRepairLine(context.Background(), file, ref, b, true))
	require.Len(t, r.upserts, 1)
	require.Len(t, r.userInputs, 1)
	require.Equal(t, "hello", r.userInputs[0].Content)
	require.Empty(t, r.states)
}

	if r.fail {
		return errors.New("injected response disk full")

func TestRepairResponseFailureCanRetryAndPreservesFacts(t *testing.T) {
	b := []byte(`{"archive_id":"repair-1","event":"response","timestamp":"2026-10-31T01:00:02Z","status":200,"response_id":"resp1","usage":{"output_tokens":42}}` + "\n")
	r := &repairResponseFailureRepo{fail: true}
	s := NewTokenAnalysisService(r, nil, nil)
	ref := ArchiveRepairLine{ArchiveID: "repair-1", Event: "response", End: int64(len(b))}
	err := s.ApplyRepairLine(context.Background(), "2026-10-31.jsonl", ref, b, false)
	require.ErrorContains(t, err, "disk full")
	require.Empty(t, r.responses)
	require.Empty(t, r.states)
	r.fail = false
	require.NoError(t, s.ApplyRepairLine(context.Background(), "2026-10-31.jsonl", ref, b, false))
	require.Len(t, r.responses, 1)
	require.Equal(t, int64(42), *r.responses[0].OutputTokens)
	require.Equal(t, "resp1", r.responses[0].ResponseID)
	require.Empty(t, r.states)
}

// Characterizes the deployed defect; permanent fix must reverse the offset assertion.
func TestRepairCharacterizesDeployedIndexerSkippingFailedResponse(t *testing.T) {
	b := []byte(`{"archive_id":"repair-1","event":"response","timestamp":"2026-10-31T01:00:02Z","status":200}` + "\n")
	file := filepath.Join(t.TempDir(), "2026-10-31.jsonl")
	require.NoError(t, os.WriteFile(file, b, 0600))
	r := &repairResponseFailureRepo{fail: true}
	s := NewTokenAnalysisService(r, nil, nil)
	result, err := s.indexArchiveFile(context.Background(), file, nil)
	require.NoError(t, err)
	require.Equal(t, int64(1), result.FailedRows)
	require.Equal(t, int64(len(b)), r.states[len(r.states)-1].LastOffset)
	require.Empty(t, r.responses)
}

func TestRepairLinksOnlySelectedRows(t *testing.T) {
	r := &tokenAnalysisRepoStub{linkCandidates: []TokenAnalysisUsageLinkCandidate{
		{ID: 1, ArchiveID: "untouched", ClientRequestID: "cid-other"},
		{ID: 2, ArchiveID: "restored", ClientRequestID: "cid-restored"},
	}}
	s := NewTokenAnalysisService(r, nil, nil)
	require.NoError(t, s.LinkRepairRows(context.Background(), map[string]bool{"restored": true}, time.Now().Add(-time.Hour), time.Now()))
	require.Len(t, r.usageLookups, 1)
	require.Equal(t, "restored", r.usageLookups[0].ArchiveID)
	require.Contains(t, r.links, "restored")
	require.NotContains(t, r.links, "untouched")
}
