package service

import (
    "context"
    "io"
    "net/http"
    "strings"
    "testing"
    "github.com/stretchr/testify/require"
    "github.com/tidwall/gjson"
)

func TestCodexRereviewPromptHashMustUseBuilderFinalBody(t *testing.T) {
    for _, passthrough:=range []bool{false,true} {
        name:="standard";if passthrough{name="passthrough"}
        t.Run(name,func(t *testing.T){
            const secret="synthetic-review-secret"
            upstream:=&httpUpstreamRecorder{resp:&http.Response{StatusCode:http.StatusOK,Header:http.Header{"Content-Type":[]string{"application/json"}},Body:io.NopCloser(strings.NewReader(`{"id":"resp_image","model":"deepseek-flash","output":[],"usage":{"input_tokens":60000,"output_tokens":2,"input_tokens_details":{"cached_tokens":1024}}}`))}}
            svc:=newOpenAIImageGenerationControlTestService(upstream)
            svc.cfg.TokenAnalysis.FingerprintSecret=secret
            c,_:=newOpenAIImageGenerationControlTestContext(true,"synthetic-review-client/1.0")
            account:=openaiMappedDeepSeekResponsesImageAccount()
            account.Extra["openai_passthrough"]=passthrough
            body:=[]byte(`{"model":"deepseek-flash","stream":false,"prompt_cache_key":"same-key","input":[{"type":"function_call","call_id":"call_image","name":"view_image","arguments":"{}"},{"type":"function_call_output","call_id":"call_image","output":[{"type":"input_image","image_url":"data:image/png;base64,AQID"}]}]}`)
            _,err:=svc.Forward(context.Background(),c,account,body)
            require.NoError(t,err)
            require.True(t,gjson.GetBytes(upstream.lastBody,"input.2.content.1.url").Exists(),"builder must have lifted/aliased the image")
            facts:=TokenAnalysisForwardFactsFromContext(c)
            require.NotNil(t,facts)
            actualItems,actualRoot:=tokenAnalysisUpstreamPromptHashes(secret,c.Request.URL.Path,upstream.lastBody)
            require.NotEmpty(t,actualItems)
            require.Equal(t,actualItems,facts.UpstreamItemsHash,"recorded final prompt must match the body actually passed to HTTPUpstream")
            require.Equal(t,actualRoot,facts.UpstreamRootHash)
        })
    }
}
